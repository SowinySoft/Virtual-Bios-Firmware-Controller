\subsection{Schematic Overview}
The schematic comprises four main sheets:

\subsubsection{Sheet 1: RP2040 Core}
\begin{itemize}
    \item Power: 3.3V from USB (via AP7361C LDO, not shown in BOM above — integrated on some RP2040 modules) or external 3.3V.
    \item Clock: 12 MHz crystal (X1) for USB; internal ROSC for PIO.
    \item GPIO Allocation (30 pins):
    \begin{itemize}
        \item GPIO 0-3: SPI0 (Original flash passthrough monitor)
        \item GPIO 4-7: SPI1/QSPI (Extension flash W25Q128)
        \item GPIO 8-11: PIO0 SM0-3 (SPI Arbiter: CS, SCK, MOSI, MISO\_IN)
        \item GPIO 12: PIO0 SM2 MISO\_OUT (to mux)
        \item GPIO 13: Mux SEL control
        \item GPIO 14-15: I2C0 (EEPROM 24C02)
        \item GPIO 16-17: USB D+/D- (via USB phy)
        \item GPIO 18-19: SWD (debug)
        \item GPIO 20-21: UART0 (console)
        \item GPIO 22-25: Status LEDs / Buttons
        \item GPIO 26-29: ADC / Reserved
    \end{itemize}
\end{itemize}

\subsubsection{Sheet 2: SPI Interposer Path}
\begin{itemize}
    \item PCH Side: CS\#, SCK, MOSI, MISO, WP\#, HOLD\#, VCC, GND (SOIC-8 clip pins 1-8)
    \item Original Flash Side: Same signals, passthrough except MISO
    \item MISO Path: 
    \begin{enumerate}
        \item Original Flash MISO $\to$ 74LVC1G3157 Input B (Pin 3)
        \item RP2040 MISO\_OUT (GPIO 12) $\to$ 74LVC1G3157 Input A (Pin 6)
        \item 74LVC1G3157 SEL (GPIO 13) $\to$ Selects A (RP2040) or B (Original)
        \item 74LVC1G3157 Output (Pin 5) $\to$ PCH MISO
    \end{enumerate}
    \item WP\#, HOLD\#: Pulled high on PCH side; RP2040 drives independently for extension flash.
\end{itemize}

\subsubsection{Sheet 3: Extension Flash \& EEPROM}
\begin{itemize}
    \item W25Q128: Connected to RP2040 QSPI (SSI) interface for XIP.
    \begin{itemize}
        \item QSPI\_SCK: GPIO 4
        \item QSPI\_CS: GPIO 5
        \item QSPI\_SD0-SD3: GPIO 6-9
    \end{itemize}
    \item 24C02 EEPROM: I2C0 (GPIO 14=SDA, GPIO 15=SCL), 4.7k pull-ups.
\end{itemize}

\subsubsection{Sheet 4: USB-C \& Power}
\begin{itemize}
    \item USB-C Receptacle: CC1/CC2 with 5.1k pull-downs (UFP mode).
    \item VBUS: Ferrite bead $\to$ 10uF $\to$ AP7361C LDO $\to$ 3.3V rail.
    \item D+/D-: 22R series $\to$ RP2040 USB PHY.
    \item ESD Protection: ESD9L5.0ST5G on VBUS, D+, D-, CC1, CC2.
\end{itemize}

\subsection{PCB Layout Constraints}
\begin{itemize}
    \item \textbf{Stackup:} 4-layer (Signal/GND/PWR/Signal), 1.6mm FR4.
    \item \textbf{Impedance:} 50$\Omega$ single-ended for SPI/USB; 90$\Omega$ differential for USB D+/D-.
    \item \textbf{Trace Width:} 0.15mm (6 mil) for 50$\Omega$ on 0.2mm prepreg.
    \item \textbf{Length Matching:} SPI CLK/MOSI/MISO matched to \SI{<5}{mm}; USB D+/D- matched to \SI{<0.5}{mm}.
    \item \textbf{Decoupling:} 100nF within 1mm of each power pin; bulk caps near supply entry.
    \item \textbf{Ground:} Continuous GND plane on L2; stitching vias near signal transitions.
    \item \textbf{SOIC-8 Clip Footprint:} Custom footprint matching Pomona 5250 pinout; through-hole for mechanical stability.
\end{itemize}

\subsection{Assembly Notes}
\begin{enumerate}
    \item RP2040 (QFN-56) requires reflow with solder paste stencil; hand-soldering possible with hot air.
    \item W25Q128 SOIC-8: Standard reflow.
    \item 74LVC1G3157 SOT-363: Small; use tweezers + hot air or fine-tip iron.
    \item USB-C: Through-hole or SMT; SMT requires careful alignment.
    \item SOIC-8 Clip: Solder to PCB pads; mechanical stress relief critical.
    \item Post-assembly: Program RP2040 via SWD (GPIO 18/19) with bootloader + firmware.
\end{enumerate}
