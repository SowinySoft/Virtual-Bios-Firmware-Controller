%--- Phase 1: Identify BIOS Chip ----------------------------------------------
\subsection*{Phase 1 — Identify BIOS Chip}

\begin{lstlisting}[style=vbfcStyle, caption={Identify BIOS Chip}]
# Run as Administrator
flashrom -p ch341a_spi -R
\end{lstlisting}

\textbf{Record:}
\begin{itemize}
    \item Vendor/Model: \rule{5cm}{0.4pt} (e.g., \texttt{Macronix MX25L6406E})
    \item Capacity: \rule{5cm}{0.4pt} (expected 8 MB = 8,388,608 bytes)
    \item Package: SOIC-8, 150 mil
    \item Pin-1 orientation: \rule{5cm}{0.4pt}
\end{itemize}

This determines shadow map capacity and supported SPI commands.

%--- Phase 2: Dump Stock BIOS -------------------------------------------------
\subsection*{Phase 2 — Dump Stock BIOS (Baseline)}

\begin{lstlisting}[style=vbfcStyle, caption={Dump Stock BIOS}]
flashrom -p ch341a_spi -r E:\g41_stock_bios_YYYYMMDD.bin -c "MX25L6406E"

\textbf{Verify:}

%--- Phase 3: Prepare Modified Firmware ---------------------------------------
\subsection*{Phase 3 — Prepare Modified Firmware}

\begin{enumerate}
    \item Open \texttt{E:\textbackslash g41\_stock\_bios\_*.bin} in \textbf{UEFITool} (v0.28+).
    \item Locate target region (G41 typical layout):
    \begin{itemize}
        \item \textbf{Setup DXE driver}: GUID \texttt{899407D7-99FE-43D8-9A21-79EC328CAC21} (or search ``Setup'')
        \item \textbf{Boot logo / POST text}: in \texttt{FV\_MAIN} or \texttt{FV\_SUPV}
        \item \textbf{ME region}: G41/ICH7 usually \textbf{no ME} — skip unless present
    \end{itemize}
    \item Extract region $\to$ modify binary (example: unlock hidden menu):
    \begin{itemize}
        \item Find conditional jump suppressing ``Advanced'' tab
        \item Patch \texttt{74 XX} $\to$ \texttt{EB XX} (JZ $\to$ JMP short)
    \end{itemize}
    \item Save \textbf{full modified image} as \texttt{E:\textbackslash g41\_modified\_bios.bin}.
\end{enumerate}

%--- Phase 4: Provision VBFC --------------------------------------------------
\subsection*{Phase 4 — Provision VBFC}

\begin{lstlisting}[style=vbfcStyle, caption={Provision VBFC}]
# 1. Connect VBFC via USB-C to Windows PC
# 2. Sign modified image (version 1 = first deployment)
vbfc-host sign --input E:\g41_modified_bios.bin --output E:\g41_signed_bios.bin --version 1

# 3. Flash to VBFC extension flash (slot 0 = active)
vbfc-host flash --image E:\g41_signed_bios.bin --slot 0

# 4. Configure shadow map for 8 MB chip
#    Region 0: 0x000000-0x0FFFFF (PEI/SEC) -> original flash
#    Region 1: 0x100000-0x1FFFFF (Setup DXE) -> hybrid + patching
#    Region 2: 0x200000-0x7FFFFF (rest) -> extension flash
vbfc-host shadow add --base 0x000000 --limit 0x0FFFFF --type original --offset 0x000000
vbfc-host shadow add --base 0x100000 --limit 0x1FFFFF --type hybrid --offset 0x100000 --patch-enable
vbfc-host shadow add --base 0x200000 --limit 0x7FFFFF --type extension --offset 0x200000
vbfc-host shadow commit

# 5. Verify
vbfc-host info
vbfc-host verify --slot 0
\end{lstlisting}

\textbf{Expected \texttt{vbfc-host info} output:}
\begin{lstlisting}[style=vbfcStyle]
Firmware: 0.9.0
Flash ID: 0xEF4017 (W25Q128) or 0xC22017 (MX25L128)
Shadow Map Entries: 3
  [0] 0x000000-0x0FFFFF -> ORIGINAL @ 0x000000
  [1] 0x100000-0x1FFFFF -> HYBRID @ 0x100000 (patch enabled)
  [2] 0x200000-0x7FFFFF -> EXTENSION @ 0x200000
\end{lstlisting}

%--- Phase 5: Wire VBFC to Motherboard ----------------------------------------
\subsection*{Phase 5 — Wire VBFC to Motherboard}

\begin{table}[h]

\caption{VBFC to Motherboard Wiring}
\label{tab:wiring}
\begin{tabular}{@{}cccccl@{}}
\toprule
\textbf{VBFC Pin} & \textbf{SOIC-8 Pin} & \textbf{Clip Pin} & \textbf{Signal} & \textbf{Notes} \\
\midrule

\bottomrule
\end{tabular}
\end{table}

\textbf{Physical steps:}
\begin{enumerate}
    \item Power OFF motherboard, unplug ATX PSU cord.
    \item Align clip pin-1 marker with chip pin-1 dot ($\triangle$ or notch).
    \item Press clip firmly — verify all 8 pins contact.
    \item Connect VBFC ribbon to clip.
    \item Connect VBFC USB-C to \textbf{same Windows PC} (power + sniffer).
    \item \textbf{Double-check} clip seating before power.
\end{enumerate}

%--- Phase 6: First Boot (Sniffer Mode) ---------------------------------------
\subsection*{Phase 6 — First Boot (Sniffer Mode)}

\begin{lstlisting}[style=vbfcStyle, caption={Start Sniffer Capture}]
# Start capture BEFORE power-on
vbfc-host sniff --duration 60 --output E:\boot_trace_1.csv
\end{lstlisting}

\begin{enumerate}
    \item Flip ATX PSU switch ON.
    \item Press motherboard power button.
    \item Observe:
    \begin{itemize}
        \item POST code display (if port 80 card)
        \item Serial console (UART header, 115200 8N1)
        \item VBFC LED: should blink rapidly during SPI traffic
    \end{itemize}
    \item Let boot proceed to BIOS setup or OS loader.
    \item Stop sniffer: \texttt{Ctrl+C}
\end{enumerate}

\textbf{Quick trace check:}
\begin{lstlisting}[style=vbfcStyle]
head -30 E:\boot_trace_1.csv
\end{lstlisting}

Look for normal SPI sequence: \texttt{0x03}/\texttt{0x0B} reads, addresses 0x000000 $\to$ 0x7FFFFF, no timeouts.

\textbf{If any test fails:}
\begin{enumerate}
    \item Power off immediately (ATX switch).

    \item Set VBFC bypass jumper (JP1) $\to$ board boots stock BIOS.

%--- Appendix B: G41 BIOS Chip Pinout -----------------------------------------
\subsection*{Appendix B — Common G41 BIOS Chip Pinout (SOIC-8)}

\begin{lstlisting}[style=vbfcStyle]
    Pin 1
   +---------+
CS# |1     8| VCC
MISO|2     7| HOLD#
WP# |3     6| SCK
GND |4     5| MOSI
   +---------+
\end{lstlisting}

\begin{itemize}
    \item \textbf{CS\#} = Chip Select (active low)
    \item \textbf{MISO} = Master In Slave Out (VBFC drives this line)
    \item \textbf{WP\#} / \textbf{HOLD\#} = pulled high by VBFC when active
\end{itemize}
